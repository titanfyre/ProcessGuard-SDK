cmake_minimum_required(VERSION 3.24)
project(ProductJsonNode LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PRODUCT_BRIDGE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../ProcessGuard-SDK-C-ffi" CACHE PATH "Product C ABI")
add_subdirectory("${PRODUCT_BRIDGE_SOURCE}" bridge)
add_library(processguard_sdk_json_node MODULE native.cpp DelayLoad.cpp)
set_target_properties(processguard_sdk_json_node PROPERTIES PREFIX "" SUFFIX ".node")
target_compile_features(processguard_sdk_json_node PRIVATE cxx_std_17)
target_compile_definitions(processguard_sdk_json_node PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=processguard_sdk_json_node)
target_include_directories(processguard_sdk_json_node PRIVATE "${NODE_HEADERS}")
target_link_libraries(processguard_sdk_json_node PRIVATE ProcessGuardSDKJsonSDK "${NODE_LIBRARY}" delayimp)
target_link_options(processguard_sdk_json_node PRIVATE /DELAYLOAD:node.exe)
